<!--
=========================================================
  GITHUB ACTION FOR THE SNAKE (ORANGE ATHLETE PALETTE)
  Create the file: .github/workflows/snake.yml
  Then run it once from the "Actions" tab (Run workflow).
  It publishes the SVGs to a branch named "output".
=========================================================

name: Generate Snake Animation

on:
  schedule:
    - cron: "0 */12 * * *"
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: write

jobs:
  generate:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Generate snake SVGs
        uses: Platane/snk/svg-only@v3
        with:
          github_user_name: ${{ github.repository_owner }}
          outputs: |
            dist/github-snake.svg?color_snake=FF4500&color_dots=ebedf0,FFD1B3,FFA366,FF7A29,FF4500
            dist/github-snake-dark.svg?palette=github-dark&color_snake=FF4500&color_dots=161b22,5C2A0F,A33A0F,E04A0F,FF6B00

      - name: Push to output branch
        uses: crazy-max/ghaction-github-pages@v3.1.0
        with:
          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
